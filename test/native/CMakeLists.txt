cmake_minimum_required(VERSION 3.14)
project(plana_clipboard_tests LANGUAGES CXX)
enable_testing()
add_executable(clipboard_test clipboard_test.cpp ../../windows/runner/image_clipboard.cpp)
target_include_directories(clipboard_test PRIVATE ../../windows/runner)
target_compile_features(clipboard_test PRIVATE cxx_std_17)
target_compile_definitions(clipboard_test PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(clipboard_test PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(clipboard_test PRIVATE user32 comctl32 shell32)
add_test(NAME clipboard_test COMMAND clipboard_test)

add_executable(diagnostics_test diagnostics_test.cpp ../../windows/runner/runtime_diagnostics.cpp)
target_include_directories(diagnostics_test PRIVATE ../../windows/runner)
target_compile_features(diagnostics_test PRIVATE cxx_std_17)
target_compile_definitions(diagnostics_test PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(diagnostics_test PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(diagnostics_test PRIVATE user32 shell32 ole32 uuid)
add_test(NAME diagnostics_test COMMAND diagnostics_test)
set_tests_properties(diagnostics_test PROPERTIES TIMEOUT 90)

add_executable(proxy_assist_test proxy_assist_test.cpp ../../windows/runner/proxy_assist.cpp)
target_include_directories(proxy_assist_test PRIVATE ../../windows/runner)
target_compile_features(proxy_assist_test PRIVATE cxx_std_17)
target_compile_definitions(proxy_assist_test PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(proxy_assist_test PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(proxy_assist_test PRIVATE windowsapp iphlpapi ws2_32)
add_test(NAME proxy_assist_test COMMAND proxy_assist_test "${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/proxy_assist/settings.png")
set_tests_properties(proxy_assist_test PROPERTIES TIMEOUT 45)
add_test(NAME proxy_assist_chinese_test COMMAND proxy_assist_test "${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/proxy_assist/settings-zh.png" --chinese)
set_tests_properties(proxy_assist_chinese_test PROPERTIES TIMEOUT 45)
